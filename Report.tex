\subsection{Vocabulary}

\subsection{Complete FOL Axioms}

\begin{axiombox}[A1 -- Every cell has at least one value]
$\displaystyle \forall i\; \forall j\; \exists v\quad Val(i,j,v)$
\end{axiombox}

\begin{axiombox}[A2 -- Every cell has at most one value]
$\displaystyle \forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\bigl(Val(i,j,v_1) \land Val(i,j,v_2)\bigr) \Rightarrow v_1 = v_2$
\end{axiombox}

\begin{axiombox}[A3 -- Row uniqueness (no duplicate in same row)]
$\displaystyle \forall i\; \forall j_1\; \forall j_2\; \forall v \quad
\bigl(Val(i,j_1,v) \land Val(i,j_2,v) \land j_1 \neq j_2\bigr) \Rightarrow \bot$
\end{axiombox}

\begin{axiombox}[A4 -- Column uniqueness (no duplicate in same column)]
$\displaystyle \forall j\; \forall i_1\; \forall i_2\; \forall v \quad
\bigl(Val(i_1,j,v) \land Val(i_2,j,v) \land i_1 \neq i_2\bigr) \Rightarrow \bot$
\end{axiombox}

\begin{axiombox}[{A5 -- Value completeness (values stay within $[1,N]$)}]
$\displaystyle \forall i\; \forall j\; \forall v \quad Val(i,j,v) \Rightarrow \bigl(v = 1 \lor v = 2 \lor \cdots \lor v = N\bigr)$
\end{axiombox}

\begin{axiombox}[A6 -- Horizontal less-than constraint]
$\displaystyle \forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\bigl(LessH(i,j) \land Val(i,j,v_1) \land Val(i,j{+}1,v_2)\bigr) \Rightarrow Less(v_1, v_2)$
\end{axiombox}

\begin{axiombox}[A7 -- Horizontal greater-than constraint]
$\displaystyle \forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\bigl(GreaterH(i,j) \land Val(i,j,v_1) \land Val(i,j{+}1,v_2)\bigr) \Rightarrow Less(v_2, v_1)$
\end{axiombox}

\begin{axiombox}[A8 -- Vertical less-than constraint]
$\displaystyle \forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\bigl(LessV(i,j) \land Val(i,j,v_1) \land Val(i{+}1,j,v_2)\bigr) \Rightarrow Less(v_1, v_2)$
\end{axiombox}

\begin{axiombox}[A9 -- Vertical greater-than constraint]
$\displaystyle \forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\bigl(GreaterV(i,j) \land Val(i,j,v_1) \land Val(i{+}1,j,v_2)\bigr) \Rightarrow Less(v_2, v_1)$
\end{axiombox}

\begin{axiombox}[A10 -- Given clues are enforced]
$\displaystyle \forall i\; \forall j\; \forall v \quad Given(i,j,v) \Rightarrow Val(i,j,v)$
\end{axiombox}

\subsection{Derivation of CNF Forms (Three Non-trivial Axioms)}

\subsubsection{Axiom A1: At Least One Value per Cell}

\textbf{Original FOL sentence:}
\[
\forall i\; \forall j\; \exists v \quad Val(i,j,v)
\]

\textbf{Step 1 -- Skolemization.} The existential quantifier $\exists v$ is replaced by a Skolem function $f(i,j)$ depending on all universally quantified variables in scope:
\[
\forall i\; \forall j \quad Val\bigl(i,\; j,\; f(i,j)\bigr)
\]

\textbf{Step 2 -- Drop universal quantifiers} (standard CNF convention):
\[
Val\bigl(i,\; j,\; f(i,j)\bigr)
\]

\textbf{CNF (single unit clause):}
\[
\boxed{Val(i,\; j,\; f(i,j))}
\]

\noindent When grounded over domain $\{1,\ldots,N\}$, this produces $N^2$ ground unit clauses, one per cell.

\subsubsection{Axiom A2: At Most One Value per Cell}

\textbf{Original FOL sentence:}
\[
\forall i\; \forall j\; \forall v_1\; \forall v_2 \quad \bigl(Val(i,j,v_1) \land Val(i,j,v_2)\bigr) \Rightarrow v_1 = v_2
\]

\textbf{Step 1 -- Eliminate implication} ($\phi \Rightarrow \psi \equiv \neg\phi \lor \psi$):
\[
\forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\neg\bigl(Val(i,j,v_1) \land Val(i,j,v_2)\bigr) \lor (v_1 = v_2)
\]

\textbf{Step 2 -- Apply De Morgan:}
\[
\forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\neg Val(i,j,v_1) \lor \neg Val(i,j,v_2) \lor (v_1 = v_2)
\]

\textbf{Step 3 -- Handle equality.} We treat $v_1 = v_2$ via the contrapositive: for distinct $v_1 \neq v_2$, no cell can hold both values. Grounded for all $v_1 \neq v_2$:

\textbf{CNF (one clause per distinct value pair per cell):}
\[
\boxed{\neg Val(i,j,v_1) \lor \neg Val(i,j,v_2)}
\quad \text{for each } v_1 \neq v_2
\]

\subsubsection{Axiom A6: Horizontal Less-Than Constraint}

\textbf{Original FOL sentence:}
\[
\forall i\; \forall j\; \forall v_1\; \forall v_2 \quad
\bigl(LessH(i,j) \land Val(i,j,v_1) \land Val(i,j{+}1,v_2)\bigr) \Rightarrow Less(v_1, v_2)
\]

\textbf{Step 1 -- Eliminate implication:}
\[
\neg\bigl(LessH(i,j) \land Val(i,j,v_1) \land Val(i,j{+}1,v_2)\bigr) \lor Less(v_1, v_2)
\]

\textbf{Step 2 -- Apply De Morgan:}
\[
\neg LessH(i,j) \lor \neg Val(i,j,v_1) \lor \neg Val(i,j{+}1,v_2) \lor Less(v_1, v_2)
\]

\textbf{Step 3 -- Ground $Less(v_1, v_2)$.} Since $Less$ is a background arithmetic predicate, it evaluates to $\top$ when $v_1 < v_2$ and $\bot$ otherwise. Clauses where $v_1 \geq v_2$ become:

\textbf{CNF (for each $v_1 \geq v_2$):}
\[
\boxed{\neg LessH(i,j) \lor \neg Val(i,j,v_1) \lor \neg Val(i,j{+}1,v_2)}
\]

This clause forbids assigning $v_1$ to cell $(i,j)$ and $v_2$ to cell $(i,j+1)$ when a ``$<$'' constraint exists and $v_1 \geq v_2$.

\section{Inference Algorithms}

\subsection{Automatic KB Generation}

Given size $N$, the algorithm instantiates all FOL axioms over the concrete domain $\{1,\ldots,N\}$.

\begin{notebox}[Overview]
For each axiom schema, substitute all valid combinations of indices $(i, j, v)$ from $\{1,\ldots,N\}$ to produce a fully grounded propositional knowledge base (KB) in CNF. The KB is then fed to a SAT/resolution engine for verification.
\end{notebox}

\begin{lstlisting}[style=pseudostyle, caption={Automatic KB Generation}]
GENERATE_KB(N):
  clauses = []
  for each i, j in {1..N} x {1..N}:
    # A1: at-least-one value per cell
    clauses.add( {Val(i,j,v) : v in 1..N} )
    # A2: at-most-one value per cell
    for each v1 != v2 in {1..N}:
      clauses.add( {-Val(i,j,v1), -Val(i,j,v2)} )
    # A3: row uniqueness
    for each j1 != j2, v:
      clauses.add( {-Val(i,j1,v), -Val(i,j2,v)} )
    # A4: column uniqueness
    for each i1 != i2, v:
      clauses.add( {-Val(i1,j,v), -Val(i2,j,v)} )
    # A6/A7: horizontal inequality clauses
    if LessH(i,j) is a given fact and j < N:
      for each v1 >= v2 in {1..N}:
        clauses.add( {-Val(i,j,v1), -Val(i,j+1,v2)} )
    # A8/A9: vertical inequality clauses (similarly)
    # A10: given clues
    if Given(i,j,v) is a fact:
      clauses.add( {Val(i,j,v)} )
  return clauses
\end{lstlisting}

\subsection{Forward Chaining}

\textbf{Idea:} Starting from known facts (\textit{Given}, \textit{LessH}, \textit{GreaterH}, etc.), repeatedly apply Modus Ponens to derive new $Val(i,j,v)$ atoms until a fixed-point is reached or a contradiction is detected. Combined with AC-3 style constraint propagation to prune domains.

\begin{lstlisting}[style=pseudostyle, caption={Forward Chaining Algorithm}]
FORWARD_CHAINING(puzzle):
  # Initialize domains
  for each cell (i,j):
    domain[i][j] = {1..N}
  for each Given(i,j,v):
    domain[i][j] = {v}    # enforce given clues

  changed = True
  while changed:
    changed = False
    for each cell (i,j) with |domain[i][j]| == 1:
      assigned_val = the unique value in domain[i][j]
      # Propagate row uniqueness
      for each j' != j in same row:
        if assigned_val in domain[i][j']:
          domain[i][j'].remove(assigned_val)
          changed = True
      # Propagate column uniqueness
      for each i' != i in same column:
        if assigned_val in domain[i'][j]:
          domain[i'][j].remove(assigned_val)
          changed = True
    # Propagate inequality constraints
    for each LessH(i,j):
      # val at (i,j) < val at (i,j+1)
      min_right = min(domain[i][j+1])
      domain[i][j] = {v : v < min_right, v in domain[i][j]}
      max_left = max(domain[i][j])
      domain[i][j+1] = {v : v > max_left, v in domain[i][j+1]}
    # Repeat for GreaterH, LessV, GreaterV ...
    # Check for contradiction
    for each cell (i,j):
      if domain[i][j] is empty:
        return CONTRADICTION
  if all cells have singleton domain:
    return SOLUTION
  else:
    return BACKTRACKING_COMPLETION(domains)
\end{lstlisting}

\subsection{Backward Chaining (SLD Resolution)}

\textbf{Idea:} Implement a Prolog-like query engine. Given a goal $Val(i,j,?)$, the interpreter reduces it to sub-goals using Horn clause rules encoding Futoshiki constraints, applying depth-first SLD resolution.

\begin{lstlisting}[style=pseudostyle, caption={Backward Chaining (SLD Resolution)}]
BACKWARD_CHAIN(goal Val(i,j,?), assignment):
  if Val(i,j,_) already assigned:
    return {assignment[i][j]}
  candidates = {1..N}
  # Prune via row uniqueness rule
  for each j' in same row where Val(i,j',v') is assigned:
    candidates.remove(v')
  # Prune via column uniqueness rule
  for each i' in same column where Val(i',j,v') is assigned:
    candidates.remove(v')
  # Prune via horizontal inequality rules
  if LessH(i, j-1) and Val(i,j-1,v') assigned:
    candidates = {v : v > v', v in candidates}
  if LessH(i, j) and Val(i,j+1,v') assigned:
    candidates = {v : v < v', v in candidates}
  # (Repeat for GreaterH, LessV, GreaterV ...)
  results = {}
  for each v in candidates:
    assignment[i][j] = v
    # Recursively prove remaining unassigned cells
    sub = BACKWARD_CHAIN(next_unassigned_goal, assignment)
    if sub != FAILURE:
      results.add(v)
    assignment[i][j] = unassigned
  return results if results != {} else FAILURE
\end{lstlisting}

\subsection{A* Search}

\textbf{State:} A partial assignment of values to cells, together with the current domains after constraint propagation.

\textbf{Transition cost:} Assigning one cell increases $g$ by 1 (uniform step cost).

\textbf{Heuristic:} $h(s) = $ number of unassigned cells after running AC-3 on the current state. See Section~\ref{sec:heuristic} for admissibility proof.

\begin{lstlisting}[style=pseudostyle, caption={A* Search Algorithm}]
A_STAR(puzzle):
  start = initial_state(puzzle)
  open = priority_queue()
  open.push(start, f = 0 + h(start))
  closed = {}

  while open is not empty:
    state = open.pop_min_f()
    if state is COMPLETE_ASSIGNMENT:
      return state.assignment
    if state in closed:
      continue
    closed.add(state)

    # Pick cell using MRV heuristic (min remaining values)
    cell = argmin over unassigned cells of |domain[cell]|

    for each v in domain[cell]:
      new_state = assign(state, cell, v)
      new_state.domains = AC3_PROPAGATE(new_state)
      if not CONTRADICTION(new_state):
        g = state.g + 1
        h = count_unassigned(new_state)  # admissible h
        open.push(new_state, f = g + h)

  return NO_SOLUTION
\end{lstlisting}

\subsection{Heuristic Admissibility Proof}
\label{sec:heuristic}

\begin{notebox}[Theorem: $h(s) = |U(s)|$ is admissible]
Let $U(s)$ be the set of unassigned cells in state $s$ after AC-3 propagation. Each action assigns exactly one cell, so the true remaining cost satisfies:
\[h^*(s) \geq |U(s)| = h(s)\]
Since $h(s)$ never overestimates $h^*(s)$, it is \textbf{admissible} and A* is guaranteed to find the optimal (minimal-step) solution.
\end{notebox}

\noindent\textbf{Stronger heuristic (optional):} Count cells whose domain is reduced to size 1 after AC-3 but whose singleton value conflicts with a peer. Any such state incurs at least one additional backtrack, so $h(s)$ can be augmented by the count of domain-empty cells detected during AC-3, giving a strictly better lower bound while remaining admissible.

\subsection{Brute-Force and Backtracking}

\textbf{Brute-force:} Enumerate all $N^{N^2}$ possible assignments and test each against all constraints. Used only as a theoretical baseline.

\textbf{Backtracking:} Depth-first search with constraint checking at each step. Attempts to fill cells in row-major order; prunes branches when a partial assignment violates any rule. No heuristic guidance.

\begin{lstlisting}[style=pseudostyle, caption={Backtracking Solver}]
BACKTRACK(assignment, empty_cells):
  if empty_cells is empty:
    return assignment  # complete solution
  (i, j) = empty_cells[0]
  for v in {1..N}:
    if IS_CONSISTENT(assignment, i, j, v):
      assignment[i][j] = v
      result = BACKTRACK(assignment, empty_cells[1:])
      if result != FAILURE:
        return result
      assignment[i][j] = unassigned
  return FAILURE

IS_CONSISTENT(assignment, i, j, v):
  # Check row uniqueness
  if v in {assignment[i][j'] : j' in row, j' != j}: return False
  # Check column uniqueness
  if v in {assignment[i'][j] : i' in col, i' != i}: return False
  # Check horizontal inequalities
  # Check vertical inequalities
  return True
\end{lstlisting}

\section{Comparative Analysis}

\subsection{Forward Chaining}

Forward chaining is highly effective on well-constrained puzzles (many given clues and inequality chains), often solving $4\times4$ and $5\times5$ instances without any backtracking. Its arc-consistency propagation eliminates large swaths of the search space in polynomial time per iteration. However, on sparse $9\times9$ instances with few given cells, it may reach a fixed-point before fully determining the solution, requiring supplementary backtracking.

\begin{notebox}[Preferred when:]
Puzzle is heavily constrained (many givens or inequalities); solving grids of size up to $6\times6$.
\end{notebox}

\subsection{Backward Chaining (SLD Resolution)}

Backward chaining queries individual cells top-down, making it convenient for partial queries (\textit{``what is the value of cell $(i,j)$?''}). It avoids computing the entire solution when only a subset of values is needed. However, its depth-first nature can lead to deep recursion and redundant sub-goal evaluation on larger grids without memoization.

\begin{notebox}[Preferred when:]
Only a subset of cell values needs to be queried; small to medium puzzle sizes; Prolog-style reasoning is required.
\end{notebox}

\subsection{A* Search}

A* with the AC-3 heuristic consistently expands far fewer nodes than backtracking. The MRV (Minimum Remaining Values) cell selection policy focuses effort on the most constrained cells first, producing tight branch pruning. On $7\times7$ and $9\times9$ puzzles, A* demonstrates clear advantages in node count over backtracking, at the cost of higher memory due to the open list.

\begin{notebox}[Preferred when:]
Large puzzle sizes; correctness and optimality (in terms of steps) are required; memory is not a bottleneck.
\end{notebox}

\subsection{Brute-Force and Backtracking}

Brute-force is infeasible beyond $4\times4$. Plain backtracking serves as a practical baseline: it is simpler to implement and uses negligible memory, but expands exponentially more nodes than A* on hard instances. It remains a useful correctness oracle for verifying other solvers.

\begin{notebox}[Preferred when:]
Verification of other algorithms; simple implementation needed; only small grids.
\end{notebox}
